(a) Shell Script to Display Disk Usage
--------------------------------------

while true
do
    echo "Enter:"
    echo "1. Show disk usage in human readable form (-h)"
    echo "2. Show disk usage in MB (-m)"
    echo "3. Show inode usage (-i)"
    echo "4. Show disk usage in powers of 1000 (-H)"
    echo "5. Show filesystem type (-T)"
    echo "6. Show local filesystems only (-l)"
    echo "7. Exit"

    read ch

    case "$ch" in
        1) df -h ;;
        2) df -m ;;
        3) df -i ;;
        4) df -H ;;
        5) df -T ;;
        6) df -l ;;
        7)
            echo "Exiting..."
            break
            ;;
        *)
            echo "Invalid Choice!"
            ;;
    esac

    echo "Do you want to continue? (y/n): "
    read ans

    case "$ans" in
        y) continue ;;
        n) echo "Exiting..."; break;;
        *) echo "Invalid input, exiting..."; break;;
    esac
done


(b) Shell Script for File & Directory Permission Commands
---------------------------------------------------------

while true
do
    echo "======================================"
    echo " File Permission Management Menu"
    echo "======================================"
    echo "1. Check file permissions"
    echo "2. Set read permission"
    echo "3. Set write permission"
    echo "4. Set execute permission"
    echo "5. Remove read permission"
    echo "6. Remove write permission"
    echo "7. Remove execute permission"
    echo "8. Exit"
    echo "--------------------------------------"
    read -p "Enter your choice [1-8]: " choice

    case $choice in
        1)
            read -p "Enter filename: " file
            if [ -e "$file" ]; then
                ls -l "$file"
            else
                echo "File does not exist."
            fi
            ;;

        2)
            read -p "Enter filename: " file
            chmod u+r "$file"
            echo "Read permission granted to user for $file."
            ;;

        3)
            read -p "Enter filename: " file
            chmod u+w "$file"
            echo "Write permission granted to user for $file."
            ;;

        4)
            read -p "Enter filename: " file
            chmod u+x "$file"
            echo "Execute permission granted to user for $file."
            ;;

        5)
            read -p "Enter filename: " file
            chmod u-r "$file"
            echo "Read permission removed for user on $file."
            ;;

        6)
            read -p "Enter filename: " file
            chmod u-w "$file"
            echo "Write permission removed for user on $file."
            ;;

        7)
            read -p "Enter filename: " file
            chmod u-x "$file"
            echo "Execute permission removed for user on $file."
            ;;

        8)
            echo "Exiting... Goodbye!"
            exit 0
            ;;

        *)
            echo "Invalid choice. Please try again."
            ;;
    esac
done


(c) Shell Script to Check & Set File Permissions
------------------------------------------------

# Usage: ./set_permissions.sh <filename> <permissions>

if [ $# -ne 2 ]; then
    echo "Incorrect arguments..."
    echo "Usage: $0 <filename> <Permissions>"
    echo "Example: $0 myfile.txt 755"
    exit 1
fi

FILENAME=$1
PERMISSIONS=$2

if [ ! -e "$FILENAME" ]; then
    echo "Error: $FILENAME does not exist!"
    exit 2
fi

echo "-------------------------------------------------"
echo "Current ownership and permissions:"
ls -ld "$FILENAME"
echo "-------------------------------------------------"

CURRENT_PERM=$(stat -c "%a" "$FILENAME")

if [ "$CURRENT_PERM" -eq "$PERMISSIONS" ]; then
    echo "No changes needed. Permissions are already set to $PERMISSIONS."
else
    echo "Setting permissions to $PERMISSIONS..."
    chmod "$PERMISSIONS" "$FILENAME"
    echo "Permissions updated successfully!"
    echo "-------------------------------------------------"
    ls -ld "$FILENAME"
fi
